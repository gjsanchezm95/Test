,
p_start AS (
  SELECT TOP 1 v.x AS txt
  FROM penult
  CROSS APPLY (VALUES([Login_ID]),([User_Name]),([Login_DateTime]),([Login_Message]),
                      ([Authentication]),([IP_Address]),([Super_User_ID])) v(x)
  WHERE v.x LIKE 'Start Date:%'
),
p_end AS (
  SELECT TOP 1 v.x AS txt
  FROM penult
  CROSS APPLY (VALUES([Login_ID]),([User_Name]),([Login_DateTime]),([Login_Message]),
                      ([Authentication]),([IP_Address]),([Super_User_ID])) v(x)
  WHERE v.x LIKE 'End Date:%'
),
l_run AS (
  SELECT TOP 1 v.x AS txt
  FROM lastrow
  CROSS APPLY (VALUES([Login_ID]),([User_Name]),([Login_DateTime]),([Login_Message]),
                      ([Authentication]),([IP_Address]),([Super_User_ID])) v(x)
  WHERE v.x LIKE 'Run on:%'
),
l_created AS (
  SELECT TOP 1 v.x AS txt
  FROM lastrow
  CROSS APPLY (VALUES([Login_ID]),([User_Name]),([Login_DateTime]),([Login_Message]),
                      ([Authentication]),([IP_Address]),([Super_User_ID])) v(x)
  WHERE v.x LIKE 'Created by:%'
)
SELECT
  TRY_CONVERT(datetime2, LTRIM(RTRIM(SUBSTRING(ps.txt, CHARINDEX(':',ps.txt)+1, 4000))), 103) AS ParamStart,
  TRY_CONVERT(datetime2, LTRIM(RTRIM(SUBSTRING(pe.txt, CHARINDEX(':',pe.txt)+1, 4000))), 103) AS ParamEnd,
  TRY_CONVERT(datetime2, LTRIM(RTRIM(SUBSTRING(lr.txt, CHARINDEX(':',lr.txt)+1, 4000))), 103) AS RunOn,
  LOWER(LTRIM(RTRIM(SUBSTRING(lc.txt, CHARINDEX(':',lc.txt)+1, 4000))))                         AS CreatedBy
FROM p_start ps
CROSS JOIN p_end pe
CROSS JOIN l_run lr
CROSS JOIN l_created lc;
