import hashlib
from io import BytesIO
from azure.storage.blob import BlobServiceClient

# Supongamos que ya tienes:
# - blob_data: bytes (resultado de download_blob().readall())
# - connection_string, dest_container, dest_blob_path (p.ej. "carpeta/archivo.mrk")

def upload_marker_from_blob_bytes(
    blob_data: bytes,
    *,
    connection_string: str,
    dest_container: str,
    dest_blob_path: str,     # dónde guardar el .mrk
    encoding: str = "utf-8"
) -> str:
    # 1) Convertir a texto y emular el Trim() del C#
    text = blob_data.decode(encoding, errors="replace")
    csv_string = "\n".join(text.splitlines()).strip()

    # 2) SHA-256 en hex minúsculas (equiv. BitConverter + Replace + ToLower)
    sha256_hex = hashlib.sha256(csv_string.encode(encoding)).hexdigest()

    # 3) Contenido del marker (hash + salto de línea, como WriteLine)
    marker_bytes = (sha256_hex + "\n").encode(encoding)
    stream = BytesIO(marker_bytes)
    stream.seek(0)

    # 4) Subir a Azure Blob
    bsc = BlobServiceClient.from_connection_string(connection_string)
    marker_client = bsc.get_blob_client(container=dest_container, blob=dest_blob_path)
    marker_client.upload_blob(stream, overwrite=True, content_type="text/plain")

    return sha256_hex
